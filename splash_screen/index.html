// Jetpack Compose Splash Screen Implementation for HomePro
// File: app/src/main/java/com/homepro/app/presentation/splash/SplashScreen.kt

package com.homepro.app.presentation.splash

import androidx.compose.animation.core.*
import androidx.compose.foundation.Image
import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.material3.CircularProgressIndicator
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.draw.clip
import androidx.compose.ui.draw.drawWithContent
import androidx.compose.ui.graphics.*
import androidx.compose.ui.graphics.drawscope.clipPath
import androidx.compose.ui.layout.ContentScale
import androidx.compose.ui.res.painterResource
import androidx.compose.ui.unit.dp
import com.homepro.app.R
import kotlinx.coroutines.delay

@Composable
fun SplashScreen(
    onSplashComplete: () -> Unit = {}
) {
    // Animation progress drivers
    val transitionState = remember { MutableTransitionState(false) }
    LaunchedEffect(Unit) {
        transitionState.targetState = true
        // Delay to allow full splash animation to finish before navigating
        delay(2600)
        onSplashComplete()
    }

    val transition = updateTransition(transitionState, label = "SplashTransition")

    // 1. House Logo scale & alpha
    val houseAlpha by transition.animateFloat(
        transitionSpec = { tween(durationMillis = 600, easing = FastOutSlowInEasing) },
        label = "HouseAlpha"
    ) { state -> if (state) 1f else 0f }

    // 2. Ring reveal progress: 0f (bottom-left) -> 1f (top-right)
    val ringProgress by transition.animateFloat(
        transitionSpec = {
            tween(
                durationMillis = 1400,
                delayMillis = 400,
                easing = FastOutSlowInEasing
            )
        },
        label = "RingProgress"
    ) { state -> if (state) 1f else 0f }

    // 3. Text wipe progress: 0f (left) -> 1f (right)
    val textProgress by transition.animateFloat(
        transitionSpec = {
            tween(
                durationMillis = 1100,
                delayMillis = 800,
                easing = CubicBezierEasing(0.25f, 0.1f, 0.25f, 1f)
            )
        },
        label = "TextProgress"
    ) { state -> if (state) 1f else 0f }

    // 4. Loader fade in
    val loaderAlpha by transition.animateFloat(
        transitionSpec = {
            tween(durationMillis = 500, delayMillis = 1600)
        },
        label = "LoaderAlpha"
    ) { state -> if (state) 1f else 0f }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.White)
    ) {
        // Background wave illustration
        Image(
            painter = painterResource(id = R.drawable.splash_screen_background),
            contentDescription = null,
            modifier = Modifier.fillMaxSize(),
            contentScale = ContentScale.Crop
        )

        // Center Content Container
        Column(
            modifier = Modifier.fillMaxSize(),
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Spacer(modifier = Modifier.weight(0.55f))

            // Logo Box with House + Animated Protective Ring
            Box(
                modifier = Modifier
                    .size(260.dp)
                    .graphicsLayer {
                        alpha = houseAlpha
                    },
                contentAlignment = Alignment.Center
            ) {
                // 1. House Logo (without ring)
                Image(
                    painter = painterResource(id = R.drawable.logo_without_ring),
                    contentDescription = "HomePro Logo",
                    modifier = Modifier.fillMaxSize(),
                    contentScale = ContentScale.Fit
                )

                // 2. Protective Ring Masked Reveal along Path (Shifted for perfect alignment: +12dp Right, +2dp Down)
                Image(
                    painter = painterResource(id = R.drawable.ring),
                    contentDescription = null,
                    modifier = Modifier
                        .fillMaxSize()
                        .offset(x = 12.dp, y = 2.dp)
                        .drawWithContent {
                            // Clip canvas using animated vector path from bottom-left to top-right
                            val path = Path().apply {
                                val w = size.width
                                val h = size.height
                                // Normalized curve matching the ring geometry
                                moveTo(w * 0.09f, h * 0.67f)
                                cubicTo(
                                    w * 0.13f, h * 0.75f,
                                    w * 0.26f, h * 0.75f,
                                    w * 0.38f, h * 0.71f
                                )
                                cubicTo(
                                    w * 0.54f, h * 0.65f,
                                    w * 0.75f, h * 0.58f,
                                    w * 0.83f, h * 0.46f
                                )
                                cubicTo(
                                    w * 0.89f, h * 0.38f,
                                    w * 0.85f, h * 0.28f,
                                    w * 0.77f, h * 0.26f
                                )
                            }
                            // Draw masked content
                            clipPath(
                                path = path,
                                clipOp = ClipOp.Intersect
                            ) {
                                this@drawWithContent.drawContent()
                            }
                        },
                    contentScale = ContentScale.Fit
                )
            }

            Spacer(modifier = Modifier.height(18.dp))

            // 3. HomePro Text (Wipe Reveal from Left to Right)
            Box(
                modifier = Modifier
                    .width(260.dp)
                    .height(86.dp)
                    .drawWithContent {
                        // Reveal horizontally from left to right
                        clipRect(
                            left = 0f,
                            top = 0f,
                            right = size.width * textProgress,
                            bottom = size.height
                        ) {
                            this@drawWithContent.drawContent()
                        }
                    }
            ) {
                Image(
                    painter = painterResource(id = R.drawable.logo_text),
                    contentDescription = "HomePro",
                    modifier = Modifier.fillMaxSize(),
                    contentScale = ContentScale.Fit
                )
            }

            Spacer(modifier = Modifier.weight(0.45f))

            // 4. Modern Bottom Circular Progress Indicator
            Box(
                modifier = Modifier
                    .size(34.dp)
                    .graphicsLayer { alpha = loaderAlpha }
            ) {
                CircularProgressIndicator(
                    strokeWidth = 3.dp,
                    color = Color(0xFF0099FF),
                    modifier = Modifier.fillMaxSize()
                )
            }

            Spacer(modifier = Modifier.height(64.dp))
        }
    }
}
